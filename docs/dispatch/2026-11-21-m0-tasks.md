```ts
export type TaskStatus = "queued" | "running" | "done" | "failed" | "cancelled";
export interface TaskSpec { taskId: string; goal: string; context: string; instructions: string; constraints: string[]; acceptanceCriteria: string[]; assumptions: string[]; requiredCapabilities: string[]; preferredAgent?: string; artifacts: ("diff" | "report" | "files")[]; }
export interface CapabilityCard { id: string; displayName: string; strengths: { instructionFollowing: 1|2|3|4|5; reasoning: 1|2|3|4|5; speed: 1|2|3|4|5; costTier: "cheap"|"standard"|"premium"; domains: string[]; }; maxConcurrency: number; transports: ("headless-cli"|"acp"|"a2a")[]; }
export interface RunContext { runId: string; taskId: string; workdir: string; signal: AbortSignal; reportEvent(kind: string, message: string): void; heartbeat(): void; }
export interface RunResult { ok: boolean; summary: string; artifacts?: { kind: "diff"|"report"|"files"; path?: string; content?: string }[]; }
export interface AgentPlugin { readonly card: CapabilityCard; healthCheck(): Promise<{ ok: boolean; detail?: string }>; execute(spec: TaskSpec, ctx: RunContext): Promise<RunResult>; cancel(runId: string): Promise<void>; }
export interface TaskRecord { id: string; spec: TaskSpec; status: TaskStatus; createdAt: number; updatedAt: number; }
export interface TaskEvent { taskId: string; runId?: string; ts: number; kind: string; message: string; }
export function newTaskId(): string; // "t-" + 短 uuid
export class TaskQueue {
  constructor(dbPath: string);
  registerAgent(card: CapabilityCard): void;
  enqueue(spec: TaskSpec): TaskRecord;
  claimNext(agentId: string): { task: TaskRecord; runId: string } | null;
  completeRun(runId: string, result: RunResult): void;
  failRun(runId: string, error: string): void;
  cancelRun(runId: string): void;
  recover(): number;
  getTask(taskId: string): TaskRecord | null;
  listTasks(): TaskRecord[];
  listEvents(taskId: string): TaskEvent[];
  close(): void;
}
export class WorkerHost {
  constructor(queue: TaskQueue, plugins: AgentPlugin[], opts?: { pollIntervalMs?: number });
  start(): void;
  stop(): Promise<void>;
}
```

#### 实现要求
- 存储 better-sqlite3，表 tasks/runs/events/agents（见 docs/DESIGN.md §3），建议 WAL。dbPath 父目录不存在则创建。
- enqueue：spec 以 JSON 落库，status=queued；id 用 spec.taskId。
- claimNext(agentId)：单事务原子完成 queued→running + 创建 Run；机械匹配规则：preferredAgent === agentId，或（无 preferredAgent 且 requiredCapabilities ⊆ 该 agent card.strengths.domains）；按 createdAt FIFO（优先级/DAG 属 M2+，不做）。
- completeRun/failRun/cancelRun：迁移落库 + 写 events；Run 记录 result/error 摘要。
- recover()：把 running 重置为 queued（M0 无租约/心跳续租，heartbeat() 为 no-op），返回恢复条数，幂等。
- 所有状态迁移写 events 表。
- WorkerHost：轮询出队（默认 200ms），尊重各 card.maxConcurrency；为每个 run 提供 RunContext（AbortController、workdir=process.cwd()（M0 简化，worktree 隔离属 M1）、reportEvent 写 events、heartbeat no-op）；execute 抛错→failRun；stop()：停止轮询、abort 在途 run 并等待落定。
- 架构不变量：core 无任何 LLM 调用；core 不得硬编码任何具体 agent 名字（包括 "echo"）；不得 import plugins/*。

#### 单测（vitest，必须覆盖）
入队/出队原子性与匹配规则；complete/fail/cancel 状态迁移；recover 模拟崩溃（claim 后不 complete，用同一 db 文件新开 TaskQueue 实例 → recover → 可重新 claim）；events 记录；WorkerHost 用测试内本地 stub 插件消费（不依赖 agent-echo）；maxConcurrency 约束。
